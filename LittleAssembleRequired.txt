# Clone Rocket 68
git clone https://github.com/habedi/rocket68

# Assemble autoboot ROM
vasmm68k_mot -Fbin rom/autoboot.asm -o rom/autoboot.bin

# Create HDD and boot sector
mkdir HDD
vasmm68k_mot -Fbin rom/shell.asm -o HDD/sector_0000.bin

# Build and run
make
make run
