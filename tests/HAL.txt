HAL_MEM:
        MOVE.L  D0, D1          ; Save function code
        CMP.L   #HAL_MEM_ALLOC, D1
        BEQ     MEM_ALLOC
        CMP.L   #HAL_MEM_FREE, D1
        BEQ     MEM_FREE
        CMP.L   #HAL_MEM_CREATE, D1
        BEQ     MEM_CREATE
        CMP.L   #HAL_MEM_DESTROY, D1
        BEQ     MEM_DESTROY
        CMP.L   #HAL_MEM_ORGANIZE, D1
        BEQ     MEM_ORGANIZE
        BRA     HAL_INVALID

MEM_ALLOC:
        ; D1 = size (bytes), D2 = alignment
        ; Returns: D0 = pointer or -1 if failed
        ; Uses free list in RAM at 0x00040000
        LEA     0x00040000, A0  ; Free list head
        MOVE.L  D1, D3          ; Size to allocate
        ADD.L   #8, D3          ; + 8 bytes header
MEM_ALLOC_LOOP:
        MOVE.L  (A0), D4        ; Next block
        BEQ     MEM_ALLOC_FAIL  ; End of list
        MOVE.L  (A0)+4, D5      ; Block size
        CMP.L   D3, D5
        BCC     MEM_ALLOC_FAIL  ; Block too small
        ; Mark as allocated (flip bit 0 of size)
        OR.L    #1, D5
        MOVE.L  D5, (A0)+4
        ; Return pointer to block start
        MOVE.L  A0, D0
        ; Update free list if needed
        MOVE.L  D4, (A0)-8      ; Fix up if split
        RTS
MEM_ALLOC_FAIL:
        MOVEQ   #-1, D0
        RTS

MEM_FREE:
        ; D1 = pointer to free
        ; Add back to free list
        LEA     0x00040000, A0
        MOVE.L  D1, (A0)        ; Insert at head (simple)
        RTS

MEM_CREATE:
        ; D1 = name pointer, D2 = size
        ; Allocate and initialize
        MOVE.L  D2, D1          ; Reuse size
        BSR     MEM_ALLOC
        MOVE.L  D0, D1          ; Save pointer
        ; Zero out memory
        MOVE.L  D2, D3
        LEA     D0, A1
MEM_CREATE_LOOP:
        MOVEQ   #0, (A1)+
        DBRA    D3, MEM_CREATE_LOOP
        MOVE.L  D1, D0
        RTS

MEM_DESTROY:
        ; D1 = pointer
        BSR     MEM_FREE
        RTS

MEM_ORGANIZE:
        ; Compact free list (coalesce adjacent blocks)
        LEA     0x00040000, A0
        ; Walk list and merge
        ; (Implementation depends on free list structure)
        RTS   


HAL_IO:
        MOVE.L  D0, D1
        CMP.L   #HAL_IO_KBD_READ, D1
        BEQ     IO_KBD_READ
        CMP.L   #HAL_IO_KBD_STATUS, D1
        BEQ     IO_KBD_STATUS
        CMP.L   #HAL_IO_MOUSE_READ, D1
        BEQ     IO_MOUSE_READ
        CMP.L   #HAL_IO_MOUSE_STATUS, D1
        BEQ     IO_MOUSE_STATUS
        CMP.L   #HAL_IO_AUDIO_INIT, D1
        BEQ     IO_AUDIO_INIT
        CMP.L   #HAL_IO_AUDIO_PLAY, D1
        BEQ     IO_AUDIO_PLAY
        CMP.L   #HAL_IO_AUDIO_STOP, D1
        BEQ     IO_AUDIO_STOP
        BRA     HAL_INVALID

IO_KBD_READ:
        ; Returns: D0 = character or 0 if no input
        MOVE.B  0x00FF0000, D0  ; Read console data register
        RTS

IO_KBD_STATUS:
        ; Returns: D0 = 1 if key available, 0 if not
        MOVE.B  0x00FF0001, D0  ; Read console status
        RTS

IO_MOUSE_READ:
        ; D0 = pointer to mouse struct {x, y, buttons}
        LEA     0x00FF0020, A0  ; Mouse registers
        MOVE.W  (A0), D1        ; X position
        MOVE.W  (A0)+2, D2      ; Y position
        MOVE.B  (A0)+4, D3      ; Buttons
        MOVE.W  D1, (D0)        ; Store to user buffer
        MOVE.W  D2, (D0)+2
        MOVE.B  D3, (D0)+4
        RTS

IO_MOUSE_STATUS:
        ; Returns: D0 = 1 if movement detected
        MOVE.B  0x00FF0024, D0
        RTS

IO_AUDIO_INIT:
        ; D1 = sample rate, D2 = channels
        MOVE.W  D1, 0x00FF0030  ; Audio control register
        MOVE.B  D2, 0x00FF0032
        RTS

IO_AUDIO_PLAY:
        ; D1 = buffer pointer, D2 = length
        LEA     0x00FF0034, A0  ; Audio DMA buffer address
        MOVE.L  D1, (A0)
        MOVE.L  D2, (A0)+4
        MOVE.B  #1, 0x00FF0038  ; Start DMA
        RTS

IO_AUDIO_STOP:
        MOVE.B  #0, 0x00FF0038  ; Stop DMA
        RTS   


HAL_TIMER:
        MOVE.L  D0, D1
        CMP.L   #HAL_TIMER_INIT, D1
        BEQ     TIMER_INIT
        CMP.L   #HAL_TIMER_START, D1
        BEQ     TIMER_START
        CMP.L   #HAL_TIMER_STOP, D1
        BEQ     TIMER_STOP
        CMP.L   #HAL_TIMER_READ, D1
        BEQ     TIMER_READ
        CMP.L   #HAL_TIMER_IRQ_SET, D1
        BEQ     TIMER_IRQ_SET
        CMP.L   #HAL_TIMER_IRQ_CLR, D1
        BEQ     TIMER_IRQ_CLR
        BRA     HAL_INVALID

TIMER_INIT:
        ; D1 = period (in CPU cycles)
        MOVE.L  D1, 0x00FF0040  ; Timer period register
        MOVE.L  #0, 0x00FF0044  ; Reset counter
        RTS

TIMER_START:
        MOVE.B  #1, 0x00FF0048  ; Enable timer
        RTS

TIMER_STOP:
        MOVE.B  #0, 0x00FF0048  ; Disable timer
        RTS

TIMER_READ:
        ; Returns: D0 = elapsed cycles
        MOVE.L  0x00FF0044, D0
        RTS

TIMER_IRQ_SET:
        ; D1 = interrupt level (1-7)
        MOVE.B  D1, 0x00FF004C  ; Set IRQ level
        RTS

TIMER_IRQ_CLR:
        MOVE.B  #0, 0x00FF004C
        RTS   



HAL_DATA:
        MOVE.L  D0, D1
        CMP.L   #HAL_DATA_READ_MEM, D1
        BEQ     DATA_READ
        CMP.L   #HAL_DATA_WRITE_MEM, D1
        BEQ     DATA_WRITE
        CMP.L   #HAL_DATA_COPY, D1
        BEQ     DATA_COPY
        CMP.L   #HAL_DATA_CONVERT, D1
        BEQ     DATA_CONVERT
        BRA     HAL_INVALID

DATA_READ:
        ; D1 = source address, D2 = dest buffer, D3 = length
        MOVE.L  D1, A0
        MOVE.L  D2, A1
        MOVE.L  D3, D4
DATA_READ_LOOP:
        MOVE.B  (A0)+, (A1)+
        DBRA    D4, DATA_READ_LOOP
        RTS

DATA_WRITE:
        ; D1 = source buffer, D2 = dest address, D3 = length
        MOVE.L  D1, A0
        MOVE.L  D2, A1
        MOVE.L  D3, D4
DATA_WRITE_LOOP:
        MOVE.B  (A0)+, (A1)+
        DBRA    D4, DATA_WRITE_LOOP
        RTS

DATA_COPY:
        ; D1 = src, D2 = dst, D3 = length
        ; Handles overlapping memory (forward or backward copy)
        MOVE.L  D1, A0
        MOVE.L  D2, A1
        MOVE.L  D3, D4
        CMP.L   A0, A1
        BCS     DATA_COPY_FWD   ; dst < src, copy forward
        ; Copy backward
DATA_COPY_BWD:
        ADD.L   D3, A0
        ADD.L   D3, A1
DATA_COPY_BWD_LOOP:
        MOVE.B  -(A0), -(A1)
        DBRA    D4, DATA_COPY_BWD_LOOP
        RTS
DATA_COPY_FWD:
DATA_COPY_FWD_LOOP:
        MOVE.B  (A0)+, (A1)+
        DBRA    D4, DATA_COPY_FWD_LOOP
        RTS

DATA_CONVERT:
        ; D1 = src, D2 = dst, D3 = length, D4 = conversion type
        ; 0 = ASCII to BCD, 1 = BCD to ASCII, 2 = Hex to BCD
        ; (Implement as needed)
        RTS   



HAL_CONTEXT:
        MOVE.L  D0, D1
        CMP.L   #HAL_CTX_SWITCH, D1
        BEQ     CTX_SWITCH
        CMP.L   #HAL_CTX_SAVE, D1
        BEQ     CTX_SAVE
        CMP.L   #HAL_CTX_RESTORE, D1
        BEQ     CTX_RESTORE
        CMP.L   #HAL_CTX_CREATE_TASK, D1
        BEQ     CTX_CREATE
        CMP.L   #HAL_CTX_DELETE_TASK, D1
        BEQ     CTX_DELETE
        BRA     HAL_INVALID

; Context structure (28 bytes):
;   0: A0  4: A1  8: A2  12: A3  16: A4  20: A5  24: A6
;   28: D0  32: D1  36: D2  40: D3  44: D4  48: D5  52: D6  56: D7
;   60: SR  64: PC  68: SP

CTX_SAVE:
        ; D1 = pointer to context struct
        MOVE.L  A0, (D1)
        MOVE.L  A1, (D1)+4
        MOVE.L  A2, (D1)+8
        MOVE.L  A3, (D1)+12
        MOVE.L  A4, (D1)+16
        MOVE.L  A5, (D1)+20
        MOVE.L  A6, (D1)+24
        MOVE.L  D0, (D1)+28
        MOVE.L  D2, (D1)+32
        MOVE.L  D3, (D1)+36
        MOVE.L  D4, (D1)+40
        MOVE.L  D5, (D1)+44
        MOVE.L  D6, (D1)+48
        MOVE.L  D7, (D1)+52
        MOVE    SR, (D1)+56
        MOVE.L  PC, (D1)+58
        RTS

CTX_RESTORE:
        ; D1 = pointer to context struct
        MOVE.L  (D1), A0
        MOVE.L  (D1)+4, A1
        MOVE.L  (D1)+8, A2
        MOVE.L  (D1)+12, A3
        MOVE.L  (D1)+16, A4
        MOVE.L  (D1)+20, A5
        MOVE.L  (D1)+24, A6
        MOVE.L  (D1)+28, D0
        MOVE.L  (D1)+32, D2
        MOVE.L  (D1)+36, D3
        MOVE.L  (D1)+40, D4
        MOVE.L  (D1)+44, D5
        MOVE.L  (D1)+48, D6
        MOVE.L  (D1)+52, D7
        MOVE    (D1)+56, SR
        JMP     (D1)+58        ; Jump to saved PC

CTX_SWITCH:
        ; D1 = current context, D2 = next context
        BSR     CTX_SAVE
        MOVE.L  D2, D1
        BSR     CTX_RESTORE
        RTS

CTX_CREATE:
        ; D1 = entry point, D2 = stack pointer
        ; Allocate context from memory pool
        RTS

CTX_DELETE:
        ; D1 = context pointer
        ; Free context
        RTS   



HAL_SYSTEM:
        MOVE.L  D0, D1
        CMP.L   #HAL_SYS_RESET, D1
        BEQ     SYS_RESET
        CMP.L   #HAL_SYS_POWER_OFF, D1
        BEQ     SYS_POWER_OFF
        CMP.L   #HAL_SYS_POWER_ON, D1
        BEQ     SYS_POWER_ON
        CMP.L   #HAL_SYS_HALT, D1
        BEQ     SYS_HALT
        BRA     HAL_INVALID

SYS_RESET:
        ; Perform warm reset: clear RAM, reload ROM vectors
        MOVE.L  #0x00080000, SP
        ; Clear RAM
        LEA     0x00080000, A0
        MOVE.L  #0x00080000/4, D0
SYS_CLEAR_LOOP:
        MOVE.L  #0, (A0)+
        DBRA    D0, SYS_CLEAR_LOOP
        ; Jump to ROM entry
        JMP     0x00000080

SYS_POWER_OFF:
        ; Set power-off flag (emulator checks this)
        MOVE.B  #1, 0x00FF00F0
        RTS

SYS_POWER_ON:
        ; Clear power-off flag
        MOVE.B  #0, 0x00FF00F0
        RTS

SYS_HALT:
        ; Halt CPU (emulator stops)
        STOP    #$2700

        MOVE.W  D2, 0x00FF0092      ; Line Y1

        MOVE.W  D2, 0x00FF00A2
        MOVE.W  D3, 0x00FF00A4
        MOVE.W  D4, 0x00FF00A6
        MOVE.B  D5, 0x00FF00A8

        RTS

        MOVE.W  D4, 0x00FF00B6

COLD_BOOT:
        ; Initialize stack pointer
        MOVE.L  #0x00080000, SP
        
        ; Initialize HAL subsystems
        ; 1. Memory manager
        LEA     0x00040000, A0
        MOVE.L  #0, (A0)          ; Free list head = NULL
        MOVE.L  #0, (A0)+4        ; No blocks yet
        
        ; 2. Timer
        MOVE.L  #0, 0x00FF0040    ; Timer period
        MOVE.L  #0, 0x00FF0044    ; Timer counter
        MOVE.B  #0, 0x00FF0048    ; Timer disabled

        MOVE.W  #800, 0x00FF0002  ; Width

        MOVE.B  D0, 0x00FF0100(D0*3+2)

        MOVE.B  #0x01, 0x00FF0010

        BTST    #0, 0x00FF0012
        BEQ     HDD_BOOT_WAIT

        MOVE.W  #0, 0x00FF0013
        MOVE.B  #0x20, 0x00FF0010
BOOT_READ_WAIT:
        BTST    #0, 0x00FF0012
        BEQ     BOOT_READ_WAIT

        LEA     0x00FF0011, A0
        LEA     0x00040000, A1
        MOVEQ   #0, D0
BOOT_COPY_LOOP:
        MOVE.B  (A0), (A1)+
        ADDQ.L  #1, D0
        CMPI    #512, D0
        BLT     BOOT_COPY_LOOP
        
        ; 8. Jump to boot code
        JMP     0x00040000
